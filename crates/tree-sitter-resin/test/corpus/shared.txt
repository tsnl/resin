==================
Inherent destruction and shared ownership
==================

struct Item { value: int,
    def drop(self: Ptr<Item>) = {};
};

def main() = {
    var owner = ArcPtr<Item> { value = 42 };
    var weak: WeakPtr<Item>;
    weak := owner.downgrade();
    var maybe: ArcPtr<Item> | None;
    maybe := weak.upgrade();
    match (maybe) {
        ArcPtr<Item>(item) => { item.value; },
        None => {},
    };
};
