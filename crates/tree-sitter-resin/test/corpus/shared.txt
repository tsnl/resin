struct Item { value: int;
    
}
fn drop(self: Ptr<Item>)  {}


fn main()  {
    let mut owner = arc_ptr_alloc::<Item>(Item { value = 42 })?;
    let mut weak: WeakPtr<Item>;
    weak = owner:downgrade();
    let mut maybe: ArcPtr<Item> | None;
    maybe = weak:upgrade();
    match (maybe) {
        ArcPtr<Item>(item) => { item:get().value; },
        None => {},
    };
}
