Immutable and mutable identifier patterns
==================
fn change(mut parameter: int, fixed: int | None) -> int {
    let initial = parameter;
    let mut result: int;
    match (fixed) {
        int(mut value) => { value = value + initial; result = value; },
        None => { result = 0; },
    };
    parameter = result;
    parameter
}
