==================
Inherent destruction and shared ownership
==================

struct Item { value: int };
impl Item {
    def drop(self: Ptr<Item>) = {};
}
def main() = {
    var owner = Arc<Item> { value = 42 };
    var weak: Weak<Item>;
    weak := owner.downgrade();
    var maybe: Arc<Item> | None;
    maybe := weak.upgrade();
    match (maybe) {
        Arc<Item>(item) => { item.value; },
        None => {},
    };
};
